# FidoNet v0.2 (Mac side) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Netmail and echomail for 68kBBS as an fsxNet leaf node — Networks DB and sysop screens, board↔echo mapping, type-2+ packet toss/scan, netmail in Mail, and an EMSI/ZMODEM poll over the modem port.

**Architecture:** Echomail *is* posts and netmail *is* mail (FTN metadata in the existing records); `ftnpkt.cla` reads/writes packets with no DB knowledge; `ftntoss.cla` moves messages between packets and the databases; `emsi.cla` runs the poll (dial → EMSI → ZMODEM send → ZMODEM receive → hang up → toss). The binkp bridge on the host is a separate plan (libftn + modem emulator); this plan ends with a host-lane e2e against a Python stand-in for it.

**Tech Stack:** Clarus (pinned `bin/clarusc` + `vendor/`), vDB, existing `zmodem.cla`; host-lane tests via `scripts/test.sh`; Python 3 + `lrzsz` + `socat` for the e2e only.

**Spec:** `docs/fidonet.md` (design, record layouts, protocol rules). Read it first; every task below cites its section.

## Global Constraints

- Clarus source is MacRoman/ASCII; `\n` emits CR; `var` declarations only at the top of a body; no `+=`/`++`; strings are ≤255-byte Pascal strings, `text` is unbounded; `string(n)` renders an int in decimal; build long strings in pieces (the 68k backend caps string temps per statement).
- Check: `bin/clarusc --rtdir vendor/runtime/clarus/ bbs.cla`. Tests: `scripts/test.sh` (every `tests/*.cla`, host lane, run from a scratch dir). 68k build: `scripts/build.sh`. Never use the old `./clarus`.
- `bbs.cla` (UI, timers) only builds for the Mac; anything that must be unit-tested lives in an includable module with no UI calls.
- `now()` is negative in 2026; never use `< 0` as "unset" on a timestamp.
- Record formats are not frozen: `Boards.*`, `Mail.*`, `BRD*`, `Networks.*` on test images are deleted and recreated, no migration code.
- Filesystem API (shipped): `file.exists(p)`, `file.info(p): FileInfo` (`size`, `isDir`, …), `file.makeDir(p)` (one level, parent must exist, existing folder fails), `file.delete(p)` (fails on an open file on the Mac), `file.list(p, names)` (files *and* folders), `file.move(p, dir)`, `f.setSize(n)`.
- Commit messages: one sentence, no AI co-authorship trailers.
- Deliberate shortcuts get a `// ponytail:` comment naming the ceiling.
- Every task ends with `scripts/test.sh` passing and `bin/clarusc --rtdir vendor/runtime/clarus/ bbs.cla` clean.

```clarus
// Copyright 2026, Andrew C. Young <andrew@vaelen.org>
// SPDX-License-Identifier: MIT

// FTN addresses (FRL-1028): zone:net/node[.point]. Pure -- parsing,
// formatting and the 8-byte big-endian form vDB records store.

include "btree.cla"   // btAppendWord

record FtnAddress {
    zone: int
    net: int
    node: int
    point: int
}

func addressIsZero(a: FtnAddress): bool {
    return a.zone == 0 and a.net == 0 and a.node == 0 and a.point == 0
}

func addressEq(a: FtnAddress, b: FtnAddress): bool {
    return a.zone == b.zone and a.net == b.net and a.node == b.node and a.point == b.point
}

// "21:1/141" or "21:1/141.5" (the point only when nonzero).
func addressStr(a: FtnAddress): string {
    var s: string = string(a.zone) + ":" + string(a.net) + "/" + string(a.node)
    if a.point != 0 { s = s + "." + string(a.point) }
    return s
}

// "21:1/141" always -- INTL and packet headers carry no point.
func address3D(a: FtnAddress): string {
    return string(a.zone) + ":" + string(a.net) + "/" + string(a.node)
}

// "1/141" -- the 2D form SEEN-BY and PATH use.
func netNodeStr(a: FtnAddress): string {
    return string(a.net) + "/" + string(a.node)
}

// All-digit string as an int; -1 if empty or any non-digit.
func digitsVal(s: string): int {
    var v: int = 0
    var i: int = 0
    if s.length == 0 { return -1 }
    while i < s.length {
        if s[i] < '0' or s[i] > '9' { return -1 }
        v = v * 10 + int(s[i]) - int('0')
        i = i + 1
    }
    return v
}

// zone:net/node[.point]; a zero record when malformed. Zone and net
// must be 1-65535, node and point 0-65535.
func parseAddress(s: string): FtnAddress {
    var a: FtnAddress
    var none: FtnAddress
    var colon: int = s.indexOf(':')
    var slash: int = s.indexOf('/')
    var dot: int = s.indexOf('.')
    if colon < 1 or slash < colon + 2 { return none }
    a.zone = digitsVal(s[0, colon])
    a.net = digitsVal(s[colon + 1, slash - colon - 1])
    if dot > slash {
        a.node = digitsVal(s[slash + 1, dot - slash - 1])
        a.point = digitsVal(s[dot + 1, s.length - dot - 1])
    } else {
        a.node = digitsVal(s[slash + 1, s.length - slash - 1])
        a.point = 0
    }
    if a.zone < 1 or a.net < 1 or a.node < 0 or a.point < 0 { return none }
    if a.zone > 65535 or a.net > 65535 or a.node > 65535 or a.point > 65535 { return none }
    return a
}

// Append the 8-byte record form: zone, net, node, point as big-endian words.
func packAddress(t: text, a: FtnAddress) {
    btAppendWord(t, a.zone)
    btAppendWord(t, a.net)
    btAppendWord(t, a.node)
    btAppendWord(t, a.point)
}

func addressAt(rec: text, off: int): FtnAddress {
    var a: FtnAddress
    a.zone = rec.wordAt(off)
    a.net = rec.wordAt(off + 2)
    a.node = rec.wordAt(off + 4)
    a.point = rec.wordAt(off + 6)
    return a
}
```

```clarus
// Copyright 2026, Andrew C. Young <andrew@vaelen.org>
// SPDX-License-Identifier: MIT

// The Networks database: one record per FTN network we are a leaf
// node on (docs/fidonet.md, "Networks"). 256-byte records, vDB record
// ID = network ID:
//   0    name (32-byte Pascal)         32   domain (32-byte Pascal)
//   64   our address (4 x u16)         72   uplink address (4 x u16)
//   80   session password (16)         96   packet password (8)
//   104  AreaFix password (16)         120  AreaFix robot name (32)
//   152  dial string (32)              184  UTC offset, minutes (i16)
//   186  poll interval, minutes (u16)  188  flags (u16; bit 0 enabled)
//   190  lastPoll (i32)                194  lastResult (u8)

include "vdb.cla"
include "ftnaddr.cla"

const networkRecordSize: int = 256
const nOffName: int = 0
const nOffDomain: int = 32
const nOffAddr: int = 64
const nOffUplink: int = 72
const nOffSessionPw: int = 80
const nOffPacketPw: int = 96
const nOffAreafixPw: int = 104
const nOffAreafixName: int = 120
const nOffDial: int = 152
const nOffUtc: int = 184
const nOffInterval: int = 186
const nOffFlags: int = 188
const nOffLastPoll: int = 190
const nOffLastResult: int = 194

const netFlagEnabled: int = 1

const pollNever: int = 0
const pollOk: int = 1
const pollNoCarrier: int = 2
const pollNotMailer: int = 3
const pollBadPassword: int = 4
const pollAddressMismatch: int = 5
const pollTransferFailed: int = 6
const pollTimeout: int = 7

record Network {
    id: int
    name: string(31)
    domain: string(31)
    addr: FtnAddress
    uplink: FtnAddress
    sessionPw: string(15)
    packetPw: string(7)
    areafixPw: string(15)
    areafixName: string(31)
    dial: string(31)
    utcOffset: int      // minutes east of UTC
    pollInterval: int   // minutes; 0 = manual only
    flags: int
    lastPoll: int       // Mac-epoch seconds; 0 = never
    lastResult: int     // poll* code
}

var networksDb: Database
var network: Network    // last loaded network

func networksOpen(): bool {
    networksDb = dbOpen("Networks", "68BB")
    if dbIsOpen(networksDb) { return true }
    networksDb = dbCreate("Networks", networkRecordSize, "68BB")
    return dbIsOpen(networksDb)
}

func networksClose() {
    if dbIsOpen(networksDb) { dbClose(networksDb) }
    networksDb = new Database
}

func networkCount(): int {
    return dbRecordCount(networksDb)
}

// For enumerating networks: loop loadNetwork over 1 .. networksNextId()-1.
func networksNextId(): int {
    return dbNextRecordId(networksDb)
}

// Zero-pad t to `off`, then append s as a Pascal field (clamped to max).
func nwPascalAt(t: text, off: int, s: string, max: int) {
    var v: string = s
    if v.length > max { v = v[0, max] }
    while t.length < off {
        t.append(char(0))
    }
    t.append(char(v.length))
    t.append(v)
}

func packNetwork(n: Network): text {
    var t: text
    nwPascalAt(t, nOffName, n.name, 31)
    nwPascalAt(t, nOffDomain, n.domain, 31)
    while t.length < nOffAddr {
        t.append(char(0))
    }
    packAddress(t, n.addr)
    packAddress(t, n.uplink)
    nwPascalAt(t, nOffSessionPw, n.sessionPw, 15)
    nwPascalAt(t, nOffPacketPw, n.packetPw, 7)
    nwPascalAt(t, nOffAreafixPw, n.areafixPw, 15)
    nwPascalAt(t, nOffAreafixName, n.areafixName, 31)
    nwPascalAt(t, nOffDial, n.dial, 31)
    while t.length < nOffUtc {
        t.append(char(0))
    }
    btAppendWord(t, n.utcOffset & 0xFFFF)
    btAppendWord(t, n.pollInterval)
    btAppendWord(t, n.flags)
    btAppendInt(t, n.lastPoll)
    t.append(char(n.lastResult))
    return t
}

// Returns the new network ID, or -1.
func createNetwork(n: Network): int {
    return dbAdd(networksDb, packNetwork(n))
}

// Write the global `network` back to its record.
func saveNetwork(): bool {
    if network.id < 1 { return false }
    return dbUpdate(networksDb, network.id, packNetwork(network))
}

func loadNetwork(id: int): bool {
    var rec: text = dbFind(networksDb, id)
    var u: int
    if rec.length != networkRecordSize { return false }
    network.id = id
    network.name = dbExtractString(rec, nOffName)
    network.domain = dbExtractString(rec, nOffDomain)
    network.addr = addressAt(rec, nOffAddr)
    network.uplink = addressAt(rec, nOffUplink)
    network.sessionPw = dbExtractString(rec, nOffSessionPw)
    network.packetPw = dbExtractString(rec, nOffPacketPw)
    network.areafixPw = dbExtractString(rec, nOffAreafixPw)
    network.areafixName = dbExtractString(rec, nOffAreafixName)
    network.dial = dbExtractString(rec, nOffDial)
    u = rec.wordAt(nOffUtc)
    if u >= 32768 { u = u - 65536 }
    network.utcOffset = u
    network.pollInterval = rec.wordAt(nOffInterval)
    network.flags = rec.wordAt(nOffFlags)
    network.lastPoll = rec.intAt(nOffLastPoll)
    network.lastResult = int(rec[nOffLastResult])
    return true
}

func networkEnabled(): bool {
    return (network.flags & netFlagEnabled) != 0
}

// First enabled network whose address is in `zone`, loaded; 0 if none.
// ponytail: linear scan -- a board has a handful of networks.
func findNetworkByZone(zone: int): int {
    var id: int = 1
    var last: int = networksNextId()
    while id < last {
        if loadNetwork(id) and networkEnabled() and network.addr.zone == zone { return id }
        id = id + 1
    }
    return 0
}

// Network by name (case-insensitive), loaded; 0 if none.
func findNetworkByName(name: string): int {
    var id: int = 1
    var last: int = networksNextId()
    while id < last {
        if loadNetwork(id) and dbStrEqCI(network.name, name) { return id }
        id = id + 1
    }
    return 0
}

func pollResultName(r: int): string {
    switch r {
    case pollNever { return "never polled" }
    case pollOk { return "ok" }
    case pollNoCarrier { return "no carrier" }
    case pollNotMailer { return "not a mailer" }
    case pollBadPassword { return "bad password" }
    case pollAddressMismatch { return "address mismatch" }
    case pollTransferFailed { return "transfer failed" }
    case pollTimeout { return "timeout" }
    }
    return "unknown"
}
```

```clarus
// board ID) -- docs/fidonet.md, "Boards":

//   64   description, 64-byte Pascal field
```

```clarus
func sendMailFtn(fromName: string, fromUserId: int, fromAddr: FtnAddress, toName: string, toUserId: int, toAddr: FtnAddress, subject: string, body: text, msgidCrc: int, created: int): int {
```

```clarus
// SPDX-License-Identifier: MIT

// FTN message text conventions and type-2+ packets (docs/fidonet.md:
// "Message text conventions", "MSGID", "Packets"). Pure -- no
// database knowledge; toss/scan live in ftntoss.cla.

include "ftnaddr.cla"
include "toolbox/osutils.cla"

const ftnCR: char = '\x0D'
const ftnLF: char = '\x0A'
const ftnSoftCR: char = '\x8D'
const ftnKludge: char = '\x01'
const ftnHexDigits: string = "0123456789ABCDEF"

func hex8(n: int): string {
    var s: string
    var i: int = 28
    while i >= 0 {
        s = s + ftnHexDigits[(n >> i) & 15]
        i = i - 4
    }
    return s
}

func hex4(n: int): string {
    var s: string
    var i: int = 12
    while i >= 0 {
        s = s + ftnHexDigits[(n >> i) & 15]
        i = i - 4
    }
    return s
}

// Hex string (either case) to int; -1 on a non-hex character or "".
func hexVal(s: string): int {
    var v: int = 0
    var i: int = 0
    var c: char
    if s.length == 0 { return -1 }
    while i < s.length {
        c = s[i]
        if c >= '0' and c <= '9' {
            v = (v << 4) | (int(c) - int('0'))
        } else if c >= 'a' and c <= 'f' {
            v = (v << 4) | (int(c) - int('a') + 10)
        } else if c >= 'A' and c <= 'F' {
            v = (v << 4) | (int(c) - int('A') + 10)
        } else {
            return -1
        }
        i = i + 1
    }
    return v
}

// --- MSGID (FTS-0009) ----------------------------------------------------

var ftnLastSerial: int = 0

// Eight hex digits, unique per origin: now() bumped past the last one
// handed out, so two messages in one second still differ.
func ftnSerial(): string {
    var n: int = now()
    if ftnLastSerial != 0 and n - ftnLastSerial <= 0 { n = ftnLastSerial + 1 }
    ftnLastSerial = n
    return hex8(n)
}

func newMsgId(a: FtnAddress): string {
    return addressStr(a) + " " + ftnSerial()
}

// CRC-32 of the MSGID text (init/final 0xFFFFFFFF); 0 for none.
func msgIdCrc(msgid: string): int {
    var t: text = msgid
    if msgid.length == 0 { return 0 }
    return t.crc32(0xFFFFFFFF, 0, t.length) ^ 0xFFFFFFFF
}

// --- Lines and kludges -----------------------------------------------------

func ftnTrim(s: string): string {
    var a: int = 0
    var b: int = s.length
    while a < b and s[a] == ' ' { a = a + 1 }
    while b > a and s[b - 1] == ' ' { b = b - 1 }
    return s[a, b - a]
}

var ftnLineEnd: int = 0     // set by ftnLine: index just past the line's CR

// The line starting at pos, without its CR.
func ftnLine(body: text, pos: int): text {
    var out: text
    var i: int = pos
    while i < body.length and body[i] != ftnCR {
        out.append(body[i])
        i = i + 1
    }
    if i < body.length { i = i + 1 }
    ftnLineEnd = i
    return out
}

// "" unless line is "^A<name>..."; else the rest, trimmed and clamped
// to 255 bytes. name includes its colon ("MSGID:") or trailing space
// ("INTL ").
func kludgeValue(line: text, name: string): string {
    var n: int = name.length
    var rest: int
    if line.length < n + 1 or line[0] != ftnKludge { return "" }
    if line[1, n] != name { return "" }
    rest = line.length - n - 1
    if rest > 255 { rest = 255 }
    return ftnTrim(line[n + 1, rest])
}

func findKludge(body: text, name: string): string {
    var pos: int = 0
    var v: string
    var line: text
    while pos < body.length {
        line = ftnLine(body, pos)
        pos = ftnLineEnd
        v = kludgeValue(line, name)
        if v != "" { return v }
    }
    return ""
}

// The reader's view: every ^A line dropped.
func stripKludges(body: text): text {
    var out: text
    var pos: int = 0
    var line: text
    while pos < body.length {
        line = ftnLine(body, pos)
        pos = ftnLineEnd
        if line.length > 0 and line[0] == ftnKludge { continue }
        out.append(line)
        out.append(ftnCR)
    }
    return out
}

// v0.2 charset rule: nothing above 0x7F leaves or enters as itself.
func ftnAscii(t: text) {
    var i: int = 0
    while i < t.length {
        if int(t[i]) >= 128 { t[i] = '?' }
        i = i + 1
    }
}
```

```clarus
const ftnMonths: string = "JanFebMarAprMayJunJulAugSepOctNovDec"

func ftnTwo(v: int): string {
    if v < 10 { return "0" + string(v) }
    return string(v)
}

func ftnDateStr(t: int): string {
    var rec: DateTimeRec
    var s: string
    SecondsToDate(t, rec)
    s = ftnTwo(rec.day) + " " + ftnMonths[(rec.month - 1) * 3, 3] + " " + ftnTwo(rec.year mod 100)
    s = s + "  " + ftnTwo(rec.hour) + ":" + ftnTwo(rec.minute) + ":" + ftnTwo(rec.second)
    return s
}

// 1-12, or 0 for an unknown month name (case-sensitive, as written).
func ftnMonthIndex(m: string): int {
    var i: int = 0
    while i < 12 {
        if ftnMonths[i * 3, 3] == m { return i + 1 }
        i = i + 1
    }
    return 0
}

// Mac-epoch seconds for "dd Mon yy  hh:mm:ss" (local time as written);
// 0 if malformed. Two-digit years: 00-79 are 20xx (FRL-1011).
func ftnDateParse(s: string): int {
    var rec: DateTimeRec
    var m: int
    var yy: int
    var d: int
    var h: int
    var mi: int
    var sec: int
    if s.length < 19 { return 0 }
    d = digitsVal(ftnTrim(s[0, 2]))
    m = ftnMonthIndex(s[3, 3])
    yy = digitsVal(s[7, 2])
    h = digitsVal(s[11, 2])
    mi = digitsVal(s[14, 2])
    sec = digitsVal(s[17, 2])
    if d < 1 or m == 0 or yy < 0 or h < 0 or mi < 0 or sec < 0 { return 0 }
    if yy < 80 { yy = yy + 2000 } else { yy = yy + 1900 }
    rec.year = yy
    rec.month = m
    rec.day = d
    rec.hour = h
    rec.minute = mi
    rec.second = sec
    rec.dayOfWeek = 1
    return DateToSeconds(rec)
}

// "-0400" / "0900" / "+0130" -> signed minutes; anything else -> 0.
func tzMinutes(tz: string): int {
    var s: string = tz
    var neg: bool = false
    var hh: int
    var mm: int
    if s.length > 0 and (s[0] == '-' or s[0] == '+') {
        neg = s[0] == '-'
        s = s[1, s.length - 1]
    }
    if s.length != 4 { return 0 }
    hh = digitsVal(s[0, 2])
    mm = digitsVal(s[2, 2])
    if hh < 0 or mm < 0 { return 0 }
    if neg { return 0 - (hh * 60 + mm) }
    return hh * 60 + mm
}

func tzStr(minutes: int): string {
    var m: int = minutes
    var s: string = ""
    if m < 0 {
        s = "-"
        m = 0 - m
    }
    return s + ftnTwo(m / 60) + ftnTwo(m mod 60)
}
```

```clarus
// Header (58 bytes, little-endian words): 0 origNode, 2 destNode,
// 4 year, 6 month (0-11), 8 day, 10 hour, 12 minute, 14 second,
// 16 baud, 18 type (2), 20 origNet, 22 destNet, 24 prodCode lo,
// 25 revision major, 26 password (8, NUL padded), 34 origZone,
// 36 destZone, 38 auxNet, 40 capValid (byte-swapped capWord),
// 42 prodCode hi, 43 revision minor, 44 capWord, 46 origZone,
// 48 destZone, 50 origPoint, 52 destPoint, 54 prodData; messages
// from 58; trailer 00 00. Packed message: 0 type (2), 2 origNode,

func pktZString(t: text, off: int, max: int): string {
    var s: string
    var i: int = 0
    while i < max and t[off + i] != '\x00' {
        s = s + t[off + i]
        i = i + 1
    }

}

func pktOpenRead(path: string): bool {
    var hdr: text
    pktClose()
    pktFile = file.open(path)
    if pktFile == nil { return false }
    pktHave = true
    pktWriting = false
    if not pktFile.readAt(0, 58, hdr) or hdr.length != 58 or hdr.wordAtLE(18) != 2 {
        pktClose()
        return false
    }
    pktOrig.node = hdr.wordAtLE(0)
    pktDest.node = hdr.wordAtLE(2)
    pktOrig.net = hdr.wordAtLE(20)
    pktDest.net = hdr.wordAtLE(22)
    if hdr.wordAtLE(44) == 1 and hdr.wordAtLE(40) == 0x0100 {
        pktOrig.zone = hdr.wordAtLE(46)     // type 2+
        pktDest.zone = hdr.wordAtLE(48)
        pktOrig.point = hdr.wordAtLE(50)
        pktDest.point = hdr.wordAtLE(52)
    } else {
        pktOrig.zone = hdr.wordAtLE(34)     // stone-age type 2 (QMail zones)
        pktDest.zone = hdr.wordAtLE(36)
        pktOrig.point = 0
        pktDest.point = 0
    }
    pktPassword = pktZString(hdr, 26, 8)
    pktPos = 58
    return true
}

// The NUL-terminated field at pktPos (NUL consumed); clears pktOk at EOF.
func pktReadZ(): text {
    var out: text
    var i: int
    while pktOk {
        if not pktFile.readAt(pktPos, 256, pktChunk) or pktChunk.length == 0 {
            pktOk = false
            return out
        }
        i = pktChunk.indexOf('\x00')
        if i < 0 {
            out.append(pktChunk)
            pktPos = pktPos + pktChunk.length
        } else {
            if i > 0 { out.append(pktChunk[0, i]) }
            pktPos = pktPos + i + 1
            return out
        }
    }
    return out
}

func pktClamp(t: text, n: int): string {
    if t.length <= n { return t[0, t.length] }
    return t[0, n]
}

// Next packed message into pktMsg/pktBody; false at the trailer/EOF.
func pktNextMsg(): bool {
    var h: text
    var raw: text
    if not pktHave or pktWriting { return false }
    if not pktFile.readAt(pktPos, 34, h) or h.length < 2 { return false }
    if h.wordAtLE(0) != 2 or h.length < 34 { return false }
    pktMsg = new PktMsg
    pktMsg.orig.node = h.wordAtLE(2)
    pktMsg.dest.node = h.wordAtLE(4)
    pktMsg.orig.net = h.wordAtLE(6)
    pktMsg.dest.net = h.wordAtLE(8)
    pktMsg.attr = h.wordAtLE(10)
    pktMsg.date = pktZString(h, 14, 19)
    pktMsg.orig.zone = pktOrig.zone
    pktMsg.dest.zone = pktDest.zone
    pktPos = pktPos + 34
    pktOk = true
    pktMsg.toName = pktClamp(pktReadZ(), 35)
    pktMsg.fromName = pktClamp(pktReadZ(), 35)
    pktMsg.subject = pktClamp(pktReadZ(), 71)
    raw = pktReadZ()
    if not pktOk { return false }
    pktBody = ftnNormalize(raw)
    return true
}

// One raw line: stops at CR, drops LF and soft-CR, ASCII-fies.
func ftnRawLine(raw: text, pos: int): text {
    var out: text
    var i: int = pos
    var c: char
    while i < raw.length and raw[i] != ftnCR {
        c = raw[i]
        if c != ftnLF and c != ftnSoftCR {
            if int(c) >= 128 { c = '?' }
            out.append(c)
        }
        i = i + 1
    }
    if i < raw.length { i = i + 1 }
    ftnLineEnd = i
    return out
}

// Inbound message text -> stored body (docs/fidonet.md, "Message text
// conventions"), harvesting the addressing/identity kludges into pktMsg.
func ftnNormalize(raw: text): text {
    var out: text
    var line: text
    var pos: int = 0
    var first: bool = true
    var v: string
    var sp: int
    var a: FtnAddress
    while pos < raw.length {
        line = ftnRawLine(raw, pos)
        pos = ftnLineEnd
        if first and line.length > 5 and line[0, 5] == "AREA:" {
            pktMsg.area = ftnTrim(line[5, line.length - 5])
            first = false
            continue
        }
        first = false
        if line.length >= 8 and line[0, 8] == "SEEN-BY:" { continue }
        if line.length > 0 and line[0] == ftnKludge {
            if kludgeValue(line, "PATH:") != "" { continue }
            v = kludgeValue(line, "MSGID:")
            if v != "" { pktMsg.msgid = v }
            v = kludgeValue(line, "REPLY:")
            if v != "" { pktMsg.reply = v }
            v = kludgeValue(line, "TZUTC:")
            if v != "" { pktMsg.tzutc = v }
            v = kludgeValue(line, "INTL ")
            if v != "" {
                sp = v.indexOf(' ')
                if sp > 0 {
                    a = parseAddress(v[0, sp])
                    if not addressIsZero(a) { pktMsg.dest.zone = a.zone }
                    a = parseAddress(v[sp + 1, v.length - sp - 1])
                    if not addressIsZero(a) { pktMsg.orig.zone = a.zone }
                }
            }
            v = kludgeValue(line, "FMPT ")
            if v != "" and digitsVal(v) > 0 { pktMsg.orig.point = digitsVal(v) }
            v = kludgeValue(line, "TOPT ")
            if v != "" and digitsVal(v) > 0 { pktMsg.dest.point = digitsVal(v) }
        }
        out.append(line)
        out.append(ftnCR)
    }
    return out
}

    if pktWriting {

        z.append('\x00')
        pktFile.writeAt(pktPos, z)
        pktFile.flush()
    }
    pktFile.close()
    pktFile = nil
    pktHave = false
    pktWriting = false
}
```

```clarus
    pktWriting = true

    pktWord(h, orig.node)
    pktWord(h, dest.node)
    pktWord(h, rec.year)
    pktWord(h, rec.month - 1)
    pktWord(h, rec.day)
    pktWord(h, rec.hour)
    pktWord(h, rec.minute)
    pktWord(h, rec.second)
    pktWord(h, 0)               // baud
    pktWord(h, 2)               // packet type
    pktWord(h, orig.net)
    pktWord(h, dest.net)
    h.append('\x00')            // product code (low): unregistered
    h.append('\x00')            // revision major
    pktFixedZ(h, password, 8)
    pktWord(h, orig.zone)       // QMail zones
    pktWord(h, dest.zone)
    pktWord(h, 0)               // auxNet
    pktWord(h, 0x0100)          // capValid: capWord byte-swapped
    h.append('\x00')            // product code (high)
    h.append('\x02')            // revision minor (0.2)
    pktWord(h, 1)               // capWord: type 2+
    pktWord(h, orig.zone)
    pktWord(h, dest.zone)
    pktWord(h, orig.point)
    pktWord(h, dest.point)
    pktWord(h, 0)               // product data
    pktWord(h, 0)
    if not pktFile.writeAt(0, h) {
        pktClose()
        return false
    }
    pktPos = 58
    return true
}

// Append one packed message. body is the finished text (kludges,
// text, tearline, origin, SEEN-BY, PATH as the caller composed it);
// the AREA line is added here from m.area.
func pktAddMsg(m: PktMsg, body: text): bool {
    var t: text
    if not pktHave or not pktWriting { return false }
    pktWord(t, 2)
    pktWord(t, m.orig.node)
    pktWord(t, m.dest.node)
    pktWord(t, m.orig.net)
    pktWord(t, m.dest.net)
    pktWord(t, m.attr)
    pktWord(t, 0)               // cost
    pktFixedZ(t, m.date, 20)
    t.append(m.toName)
    t.append('\x00')
    t.append(m.fromName)
    t.append('\x00')
    t.append(m.subject)
    t.append('\x00')
    if m.area != "" {
        t.append("AREA:")
        t.append(m.area)
        t.append(ftnCR)
    }
    t.append(body)
    t.append('\x00')
    if not pktFile.writeAt(pktPos, t) { return false }
    pktPos = pktPos + t.length
    return true
}
```

```clarus
    if t == 0 { return now() }
    if pktMsg.tzutc != "" { t = t - tzMinutes(pktMsg.tzutc) * 60 + network.utcOffset * 60 }

func tossSubject(): string {
    if pktMsg.subject.length > 63 { return pktMsg.subject[0, 63] }
    return pktMsg.subject

func tossEchomail(netId: int) {
    var bid: int = boardForTag(netId, pktMsg.area)
    var crc: int
    var parent: int
    var thread: int = 0
    if bid == 0 {
        tossUnknown = tossUnknown + 1
        return
    }
    if not postsOpen(bid) {
        log("Toss: cannot open board " + string(bid) + ": " + lastError.message)
        return
    }
    crc = msgIdCrc(pktMsg.msgid)
    if findPostByMsgId(crc) > 0 {
        tossDupes = tossDupes + 1
        return
    }
    if pktMsg.reply != "" {
        parent = findPostByMsgId(msgIdCrc(pktMsg.reply))
        if parent > 0 and loadPost(parent) {
            if post.threadId == 0 { thread = parent } else { thread = post.threadId }
        }
    }
    if addPostFtn(pktMsg.fromName, tossCreated(), thread, tossSubject(), pktBody, crc, pktMsg.orig, postFlagInbound) < 1 {
        log("Toss: post add failed: " + lastError.message)
        return
    }
    tossEcho = tossEcho + 1
}

// the original name kept in toName.
func tossNetmail(netId: int) {

    if uid < 1 { uid = 1 }
    if sendMailFtn(pktMsg.fromName, 0, pktMsg.orig, pktMsg.toName, uid, pktMsg.dest, tossSubject(), pktBody, msgIdCrc(pktMsg.msgid), tossCreated()) < 1 {
        log("Toss: mail add failed: " + lastError.message)
        return
    }
    tossNet = tossNet + 1
}

// One packet: header checks, then every message. False leaves the
// file in place (logged) -- a packet we can't read or that isn't ours.
func tossFile(path: string): bool {
    var netId: int
    if not pktOpenRead(path) {
        log("Toss: not a packet: " + path)
        return false
    }
    netId = networkForPacket()
    if netId == 0 {
        log("Toss: " + path + " is for " + addressStr(pktDest) + ", not us; left in place.")
        pktClose()
        return false
    }
    if network.packetPw != "" and not dbStrEqCI(pktPassword, network.packetPw) {
        log("Toss: " + path + ": wrong packet password; left in place.")
        pktClose()
        return false
    }
    while pktNextMsg() {
        if pktMsg.area != "" {
            tossEchomail(netId)
        } else {
            tossNetmail(netId)
        }
    }
    pktClose()
    return true
}

// Every packet in :FTN:In, deleted once tossed. Returns the count.
func tossInbound(): int {
    var names: list of string
    var fi: FileInfo
    var i: int = 0
    var n: int = 0
    var p: string
    tossEcho = 0
    tossNet = 0
    tossDupes = 0
    tossUnknown = 0
    loadTagCache()
    if not file.list(ftnIn, names) {
        log("Toss: cannot list " + ftnIn + ": " + lastError.message)
        return 0
    }
    while i < names.count {
        p = ftnIn + ":" + names[i]
        fi = file.info(p)
        if not fi.isDir {
            if tossFile(p) {
                n = n + 1
                if not file.delete(p) { log("Toss: cannot delete " + p + ": " + lastError.message) }
            }
        }
        i = i + 1
    }
    postsClose()
    log("Tossed " + string(n) + " packet(s): " + string(tossEcho) + " echomail, " + string(tossNet) + " netmail, " + string(tossDupes) + " dupes, " + string(tossUnknown) + " unknown areas.")
    return n
}
```

```clarus
// The loaded outbound message into the open packet.
func scanMailMsg() {
    var m: PktMsg
    var t: text
    m.fromName = scanClamp(message.fromName, 35)
    m.toName = scanClamp(message.toName, 35)
    m.subject = scanClamp(message.subject, 71)
    m.orig = network.addr
    m.dest = message.toAddr
    m.attr = pktAttrPrivate | pktAttrLocal
    m.date = ftnDateStr(message.created)
    m.area = ""
    scanKludge(t, "INTL " + address3D(message.toAddr) + " " + address3D(network.addr))
    if message.toAddr.point != 0 { scanKludge(t, "TOPT " + string(message.toAddr.point)) }
    if network.addr.point != 0 { scanKludge(t, "FMPT " + string(network.addr.point)) }
    scanCommonKludges(t)
    t.append(messageBody(message.id))
    ftnAscii(t)
    scanTearline(t)
    if pktAddMsg(m, t) {
        scanCount = scanCount + 1
        scanMail.add(message.id)
    }
}

// Build :FTN:Out:<id>.pkt for network netId; returns the message
// count (0 = nothing pending, no file left behind).
func scanNetwork(netId: int): int {
    var ids: list of int
    var id: int = 1
    var last: int
    var i: int = 0
    scanBoards.clear()
    scanHigh.clear()
    scanMail.clear()
    scanCount = 0
    if not loadNetwork(netId) { return 0 }
    if not pktCreate(ftnOutPath(netId), network.addr, network.uplink, network.packetPw) {
        log("Scan: cannot create " + ftnOutPath(netId) + ": " + lastError.message)
        return 0
    }
    last = boardsNextId()
    while id < last {
        if loadBoard(id) and boardNetworked() and board.networkId == netId { scanBoard() }
        id = id + 1
    }
    postsClose()
    outboundMailIds(ids)
    while i < ids.count {
        if loadMessage(ids[i]) and addressEq(message.fromAddr, network.addr) { scanMailMsg() }
        i = i + 1
    }
    pktClose()
    if scanCount == 0 {
        file.delete(ftnOutPath(netId))
        scanBoards.clear()
        scanMail.clear()
    }
    return scanCount
}

// The packet was acknowledged: move the marks and drop the file.
func scanCommit(netId: int) {
    var i: int = 0
    while i < scanBoards.count {
        if loadBoard(scanBoards[i]) {
            board.lastExported = scanHigh[i]
            if not saveBoard() { log("Scan: mark save failed: " + lastError.message) }
        }
        i = i + 1
    }
    i = 0
    while i < scanMail.count {
        markSent(scanMail[i])
        i = i + 1
    }
    if file.exists(ftnOutPath(netId)) { file.delete(ftnOutPath(netId)) }
    scanBoards.clear()
    scanHigh.clear()
    scanMail.clear()
}

// Counts for the sysop's network card.
func ftnPendingPosts(netId: int): int {
    var id: int = 1
    var last: int = boardsNextId()
    var pid: int
    var plast: int
    var n: int = 0
    while id < last {
        if loadBoard(id) and boardNetworked() and board.networkId == netId and postsOpen(id) {
            pid = board.lastExported + 1
            plast = postsNextId()
            while pid < plast {
                if loadPost(pid) and (post.flags & postFlagInbound) == 0 { n = n + 1 }
                pid = pid + 1
            }
        }
        id = id + 1
    }
    postsClose()
    return n
}

func ftnPendingMail(netId: int): int {
    var ids: list of int
    var i: int = 0
    var n: int = 0
    if not loadNetwork(netId) { return 0 }
    outboundMailIds(ids)
    while i < ids.count {
        if loadMessage(ids[i]) and addressEq(message.fromAddr, network.addr) { n = n + 1 }
        i = i + 1
    }
    return n
}
```

```clarus
// Copyright 2026, Andrew C. Young <andrew@vaelen.org>
// SPDX-License-Identifier: MIT

// The poll: dial, EMSI handshake (FSC-0056, calling side), ZMODEM send
// then receive, hang up, toss (docs/fidonet.md, "Poll"). A pure state
// machine over the program's hooks: ftnModemSend (modem commands),
// xferOut (session bytes), the ZMODEM engine's xferDone/xferReceived,
// and the globals isConnected/hangupPhase. Ticks are half seconds.

include "ftntoss.cla"
include "zmodem.cla"

enum EmsiState {
    emIdle
    emDial          // ATDT sent, waiting for CONNECT
    emInit          // whacking CR, watching for EMSI_REQ
    emWaitAck       // our EMSI_DAT sent
    emWaitDat       // ACKed; waiting for theirs
    emSend          // ZMODEM send (our packet or an empty batch)
    emRecv          // ZMODEM receive into :FTN:Tmp
    emHangup        // +++/ATH under way
}

const emDialTicks: int = 120        // 60 s for CONNECT
const emInitTicks: int = 120        // 60 s for EMSI_REQ
const emInqTick: int = 40           // 20 s: send EMSI_INQ twice
const emAckTicks: int = 40          // resend EMSI_DAT every 20 s
const emDatTicks: int = 120         // 60 s for their DAT
const emSessionTicks: int = 1800    // 15 min cap
const emMaxTries: int = 6

var emState: EmsiState = emIdle
var emTimer: int = 0
var emTotal: int = 0
var emTries: int = 0
var emLine: text
var emGotByte: bool = false
var emHadFile: bool = false
var emSendOk: bool = false

// --- Requests and the queue ---------------------------------------------------

func ftnQueued(netId: int): bool {
    var i: int = 0
    while i < ftnQueue.count {
        if ftnQueue[i] == netId { return true }
        i = i + 1
    }
    return false
}

func ftnQueuePoll(netId: int) {
    if ftnPolling and netId == ftnNetId { return }
    if not ftnQueued(netId) { ftnQueue.add(netId) }
}

// Poll now if the line is idle, else after the current call/poll.
func ftnPollRequest(netId: int) {
    if ftnPolling or isConnected {
        if netId != ftnNetId or not ftnPolling { ftnQueuePoll(netId) }
        return
    }
    ftnPollStart(netId)
}

func ftnPollAll() {
    var id: int = 1
    var last: int = networksNextId()
    while id < last {
        if loadNetwork(id) and networkEnabled() { ftnQueuePoll(id) }
        id = id + 1
    }
    ftnNext()
}

// Start the next queued poll when nothing is on the line.
func ftnNext() {
    var id: int
    if ftnPolling or isConnected or ftnQueue.count == 0 { return }
    id = ftnQueue.shift()
    ftnPollStart(id)
}

// Interval polls; call once a minute.
func ftnSchedule() {
    var id: int = 1
    var last: int
    if ftnPolling or isConnected { return }
    last = networksNextId()
    while id < last {
        if loadNetwork(id) and networkEnabled() and network.pollInterval > 0 {
            if network.lastPoll == 0 or now() - network.lastPoll >= network.pollInterval * 60 { ftnQueuePoll(id) }
        }
        id = id + 1
    }
    ftnNext()
}

// --- The session -----------------------------------------------------------

func ftnPollStart(netId: int) {
    if not loadNetwork(netId) or not networkEnabled() { return }
    if not ftnEnsureFolders() {
        log("FTN: cannot create the :FTN folders: " + lastError.message)
        return
    }
    ftnPolling = true
    ftnNetId = netId
    emHadFile = scanNetwork(netId) > 0
    loadNetwork(netId)
    emWireName = hex8(now()) + ".pkt"
    emFiles = 0
    emSendOk = false
    emResult = pollNever
    emTotal = 0
    emTimer = 0
    emTries = 0
    emLine = ""
    emGotByte = false
    log("Polling " + network.name + " at " + addressStr(network.uplink) + "...")
    emState = emDial
    ftnModemSend("ATDT" + network.dial + "\x0D")
}

// CONNECT arrived (bbs.cla's connected() while ftnPolling).
func emsiConnected() {
    emState = emInit
    emTimer = 0
    emGotByte = false
    emLine = ""
}

func emSendSeq(name: string) {
    var t: text = name
    var out: text
    out.append("**")
    out.append(name)
    out.append(hex4(t.crc16x(0, 0, t.length)))
    out.append('\x0D')
    xferOut(out)
}

func emSendDat() {
    var d: text
    var p: text
    var out: text
    emTries = emTries + 1
    if emTries > emMaxTries {
        emFail(pollTimeout)
        return
    }
    p.append("{EMSI}{" + addressStr(network.addr) + "}{" + network.sessionPw + "}")
    p.append("{8N1,PUA}{ZMO,NRQ}{00}{68kBBS}{0.2}{" + ftnVersion + "}")
    p.append("{IDENT}{[" + ftnSystemName + "][][" + ftnSysopName() + "][-Unpublished-][57600][]}")
    d.append("EMSI_DAT")
    d.append(hex4(p.length))
    d.append(p)
    out.append("**")
    out.append(d)
    out.append(hex4(d.crc16x(0, 0, d.length)))
    out.append('\x0D')
    xferOut(out)

    emTimer = 0
}

// The k-th {...} field of an EMSI_DAT body (0 = "{EMSI}"), "" if absent.
func emField(data: text, k: int): string {
    var i: int = 0
    var n: int = 0
    var s: string
    var depth: int = 0
    while i < data.length {
        if data[i] == '{' {
            depth = depth + 1
            if depth == 1 and n == k { s = "" }
        } else if data[i] == '}' {
            if depth == 1 and n == k { return s }
            if depth == 1 { n = n + 1 }
            depth = depth - 1
        } else if depth >= 1 and n == k and s.length < 255 {
            s = s + data[i]
        }
        i = i + 1
    }
    return ""
}

// Their address list (space-separated) must include the uplink.
func emAddressOk(list: string): bool {
    var s: string = list
    var sp: int
    var one: string
    while s.length > 0 {
        sp = s.indexOf(' ')
        if sp < 0 {
            one = s
            s = ""
        } else {
            one = s[0, sp]
            s = s[sp + 1, s.length - sp - 1]
        }
        if addressEq(parseAddress(one), network.uplink) { return true }
    }
    return false
}

// "**EMSI_DAT<len><data><crc>" starts at emLine[at].
func emRecvDat(at: int) {
    var len: int
    var body: text
    var data: text
    if emLine.length < at + 18 { return }
    len = hexVal(emLine[at + 10, 4])
    if len < 0 or emLine.length < at + 18 + len {
        emSendSeq("EMSI_NAK")
        return
    }
    body = emLine.textAt(at + 2, 12 + len)
    if hexVal(emLine[at + 14 + len, 4]) != body.crc16x(0, 0, body.length) {
        emSendSeq("EMSI_NAK")
        return
    }
    emSendSeq("EMSI_ACK")
    emSendSeq("EMSI_ACK")
    data = emLine.textAt(at + 14, len)
    if not emAddressOk(emField(data, 1)) {
        log("EMSI: peer presented " + emField(data, 1) + ", not " + addressStr(network.uplink))
        emFail(pollAddressMismatch)
        return
    }
    if network.sessionPw != "" and not dbStrEqCI(emField(data, 2), network.sessionPw) {
        log("EMSI: session password mismatch.")
        emFail(pollBadPassword)
        return
    }
    log("EMSI: session with " + emField(data, 6) + " " + emField(data, 7))
    emState = emSend
    emTimer = 0
    if emHadFile {
        zmodemSendStart(ftnOutPath(ftnNetId), emWireName)
    } else {
        zmodemSendEmpty()
    }
}

func emLineDone() {
    var i: int = emLine.indexOf("**EMSI_")
    var name: string
    if i < 0 or emLine.length < i + 14 { return }
    name = emLine[i + 2, 8]
    switch name {
    case "EMSI_REQ" {
        if emState == emInit {
            emSendSeq("EMSI_INQ")
            emSendDat()
        } else if emState == emWaitAck {
            emSendDat()
        }
    }
    case "EMSI_ACK" {
        if emState == emWaitAck {
            emState = emWaitDat
            emTimer = 0
        }
    }
    case "EMSI_NAK" { if emState == emWaitAck { emSendDat() } }
    case "EMSI_HBT" { emTimer = 0 }
    case "EMSI_DAT" { if emState == emWaitDat { emRecvDat(i) } }
    else { }
    }
}

func emsiChar(c: char) {
    var b: char = char(int(c) & 0x7F)
    switch emState {
    case emSend, emRecv {
        zmodemChar(c)
        return
    }
    case emInit, emWaitAck, emWaitDat { }
    else { return }
    }
    emGotByte = true
    if b == '\x0D' {
        emLineDone()
        emLine = ""
        return
    }
    if emLine.length < 4096 { emLine.append(b) }
}

func emsiTick() {
    if emState == emIdle or emState == emHangup { return }
    emTotal = emTotal + 1
    emTimer = emTimer + 1
    if emTotal > emSessionTicks {
        emFail(pollTimeout)
        return
    }
    switch emState {
    case emDial { if emTimer >= emDialTicks { emFail(pollNoCarrier) } }
    case emInit {
        if emTimer >= emInitTicks {
            emFail(pollNotMailer)
        } else if emTimer == emInqTick {
            emSendSeq("EMSI_INQ")
            emSendSeq("EMSI_INQ")
        } else if not emGotByte and emTimer mod 2 == 0 {
            xferOut("\x0D")
        }
    }
    case emWaitAck { if emTimer >= emAckTicks { emSendDat() } }
    case emWaitDat { if emTimer >= emDatTicks { emFail(pollTimeout) } }
    case emSend, emRecv { zmodemTick() }
    else { }
    }
}

// The ZMODEM engine finished a direction.
func emsiXferDone(ok: bool) {
    if emState == emSend {
        emSendOk = ok
        if not ok {
            emFail(pollTransferFailed)
            return
        }
        emState = emRecv
        emTimer = 0
        zmodemRecvStart(ftnTmp)
        return
    }
    if emState == emRecv {
        if ok {
            emFinish(pollOk)
        } else {
            emFail(pollTransferFailed)
        }
    }
}

// A complete inbound file: :FTN:Tmp -> :FTN:In, where toss looks.
func emsiFileReceived(name: string, bytes: int) {
    if file.move(ftnTmp + ":" + name, ftnIn) {
        emFiles = emFiles + 1
        log("FTN: received " + name + " (" + string(bytes) + " bytes)")
    } else {
        log("FTN: cannot move " + name + " into " + ftnIn + ": " + lastError.message)
    }
}

// Wire names from the bridge: 1-31 bytes, no colon.
func emsiAcceptName(name: string): bool {
    var i: int = 0
    if name.length == 0 or name.length > 31 { return false }
    while i < name.length {
        if name[i] == ':' { return false }
        i = i + 1
    }
    return true
}

func emFail(r: int) {
    log("FTN: poll failed: " + pollResultName(r))
    emResult = r
    emHangup()
}

func emFinish(r: int) {
    emResult = r
    emHangup()
}

// Drop the line: via bbs.cla's paced +++/ATH when connected, else a
// plain ATH from command mode.
func emHangup() {
    if emState == emSend or emState == emRecv { zmodemAbort() }
    emState = emHangup
    if isConnected {
        hangupPhase = 1
    } else {
        ftnModemSend("ATH\x0D")
        emsiDisconnected()
    }
}

// Carrier lost or hang-up finished (bbs.cla's disconnected() while
// ftnPolling): toss, commit, record, then the next queued poll.
func emsiDisconnected() {
    var n: int = 0
    if emState == emIdle { return }
    if emState == emSend or emState == emRecv { zmodemAbort() }
    if emResult == pollNever {
        if emState == emDial {
            emResult = pollNoCarrier
        } else {
            emResult = pollTransferFailed
        }
    }
    emState = emIdle
    if emSendOk { scanCommit(ftnNetId) }
    n = tossInbound()
    if loadNetwork(ftnNetId) {
        network.lastPoll = now()
        network.lastResult = emResult
        if not saveNetwork() { log("FTN: poll record save failed: " + lastError.message) }
        log(network.name + ": " + pollResultName(emResult) + "; received " + string(emFiles) + " file(s), tossed " + string(n) + ".")
    }
    ftnPolling = false
    ftnNetId = 0
    ftnNext()
}

// Someone else needs the engine (a real caller's disconnect path).
func emsiAbort() {
    if emState == emSend or emState == emRecv { zmodemAbort() }
    if emState != emIdle { emState = emHangup }
}
```

```python
#!/usr/bin/env python3
"""A fake modem + answering EMSI/ZMODEM mailer for scripts/ftn-e2e.sh.

Listens on --port; the harness connects (CLARUS_SERIAL_MODEM=connect:).
Reads "ATDT<n>\r", answers CONNECT, runs the EMSI answer side, receives
the caller's batch with lrz into --inbound, sends every file in
--outbound with lsz, then answers +++/ATH with NO CARRIER.
"""
import argparse, os, socket, subprocess, sys, glob

def crc16x(data: bytes) -> int:
    c = 0
    for b in data:
        c ^= b << 8
        for _ in range(8):
            c = ((c << 1) ^ 0x1021) & 0xFFFF if c & 0x8000 else (c << 1) & 0xFFFF
    return c

def seq(name: bytes) -> bytes:
    return b"**" + name + b"%04X" % crc16x(name) + b"\r"

def dat(fields: bytes) -> bytes:
    body = b"EMSI_DAT" + b"%04X" % len(fields) + fields
    return b"**" + body + b"%04X" % crc16x(body) + b"\r"

def read_until(sock, marker, limit=8192):
    buf = b""
    while marker not in buf and len(buf) < limit:
        chunk = sock.recv(1)
        if not chunk:
            raise EOFError
        buf += chunk
    return buf

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--port", type=int, required=True)
    ap.add_argument("--address", default="21:1/100")
    ap.add_argument("--password", default="SECRET")
    ap.add_argument("--inbound", required=True)
    ap.add_argument("--outbound", required=True)
    a = ap.parse_args()
    ls = socket.socket()
    ls.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    ls.bind(("127.0.0.1", a.port)); ls.listen(1)
    sock, _ = ls.accept()
    print("peer: harness connected", flush=True)
    line = read_until(sock, b"\r")
    if not line.startswith(b"ATDT"):
        print("peer: expected ATDT, got %r" % line); sys.exit(1)
    sock.sendall(b"\r\nCONNECT 57600\r\n")
    sock.sendall(seq(b"EMSI_REQ"))
    got = read_until(sock, b"**EMSI_INQ")
    read_until(sock, b"\r")                       # the INQ's crc+CR
    theirs = read_until(sock, b"\r")              # their EMSI_DAT
    if not theirs.startswith(b"**EMSI_DAT"):
        print("peer: expected EMSI_DAT, got %r" % theirs); sys.exit(1)
    n = int(theirs[10:14], 16)
    body = theirs[2:14 + n]
    if int(theirs[14 + n:18 + n], 16) != crc16x(body):
        print("peer: bad DAT crc"); sys.exit(1)
    fields = theirs[14:14 + n].split(b"}")
    addr, pw = fields[1][1:], fields[2][1:]
    print("peer: caller %s password %s" % (addr.decode(), pw.decode()), flush=True)
    if pw.decode() != a.password:
        print("peer: password mismatch"); sys.exit(1)
    sock.sendall(seq(b"EMSI_ACK") + seq(b"EMSI_ACK"))
    sock.sendall(dat(b"{EMSI}{%s}{%s}{8N1,PUA}{ZMO,NRQ}{FE}{emsi-peer}{0.1}{}" % (a.address.encode(), a.password.encode())))
    read_until(sock, b"**EMSI_ACK"); read_until(sock, b"\r")
    fd = sock.fileno()
    r = subprocess.run(["lrz", "--zmodem", "-b", "-q", "-y"], cwd=a.inbound, stdin=fd, stdout=fd, stderr=subprocess.DEVNULL)
    print("peer: lrz exit %d, inbound %s" % (r.returncode, sorted(os.listdir(a.inbound))), flush=True)
    files = sorted(glob.glob(os.path.join(a.outbound, "*.pkt")))
    cmd = ["lsz", "--zmodem", "-b", "-q"] + files
    r = subprocess.run(cmd, stdin=fd, stdout=fd, stderr=subprocess.DEVNULL)
    print("peer: lsz exit %d, sent %s" % (r.returncode, [os.path.basename(f) for f in files]), flush=True)
    tail = b""
    while b"ATH" not in tail:
        chunk = sock.recv(64)
        if not chunk:
            break
        tail += chunk
    sock.sendall(b"\r\nNO CARRIER\r\n")
    sock.close()
    print("peer: hung up", flush=True)

if __name__ == "__main__":
    main()
```
